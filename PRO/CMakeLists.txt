cmake_minimum_required(VERSION 3.24)
project(JerzyBassAmbientPRO VERSION 0.7.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
add_executable(BassAmbientTests tests/CoreTests.cpp)
target_include_directories(BassAmbientTests PRIVATE Source)
add_test(NAME BassAmbientCore COMMAND BassAmbientTests)
find_package(Threads REQUIRED)
add_executable(BassAmbientProTests tests/ProTests.cpp)
target_include_directories(BassAmbientProTests PRIVATE Source)
target_link_libraries(BassAmbientProTests PRIVATE Threads::Threads)
add_test(NAME BassAmbientPRO COMMAND BassAmbientProTests)
add_executable(BassAmbientComposerTests tests/ComposerTests.cpp)
target_include_directories(BassAmbientComposerTests PRIVATE Source)
add_test(NAME BassAmbientComposer COMMAND BassAmbientComposerTests)
add_executable(BassAmbientRender tools/RenderDemo.cpp)
target_include_directories(BassAmbientRender PRIVATE Source)
option(BASS_AMBIENT_CORE_ONLY "Build dependency-free DSP tests only" OFF)
if(BASS_AMBIENT_CORE_ONLY)
  return()
endif()
option(PRO_ENABLE_LTO "Enable link-time optimisation for distribution builds" ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 9.0.3 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(JerzyBassAmbientPRO
  COMPANY_NAME "Jerzy" BUNDLE_ID "pl.jerzy.bassambient.pro"
  PLUGIN_MANUFACTURER_CODE Jrzy PLUGIN_CODE Jbpr
  FORMATS VST3 Standalone PRODUCT_NAME "Jerzy Bass Ambient PRO"
  IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
  EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(JerzyBassAmbientPRO)
target_sources(JerzyBassAmbientPRO PRIVATE Source/PluginProcessor.cpp Source/ComposerProcessor.cpp Source/ComposerView.cpp Source/AudioRender.cpp Source/PluginEditor.cpp Source/JerzyVSTGuiKit.h)
target_compile_options(JerzyBassAmbientPRO PUBLIC $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_compile_definitions(JerzyBassAmbientPRO PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(JerzyBassAmbientPRO PRIVATE Threads::Threads juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
if(PRO_ENABLE_LTO)
  target_link_libraries(JerzyBassAmbientPRO PUBLIC juce::juce_recommended_lto_flags)
endif()
if(BUILD_TESTING)
  juce_add_console_app(BassAmbientIntegration PRODUCT_NAME "Bass Ambient Integration Tests")
  juce_generate_juce_header(BassAmbientIntegration)
  target_sources(BassAmbientIntegration PRIVATE tests/IntegrationTests.cpp)
  target_include_directories(BassAmbientIntegration PRIVATE Source)
  target_link_libraries(BassAmbientIntegration PRIVATE JerzyBassAmbientPRO juce::juce_audio_utils)
  add_test(NAME BassAmbientHostGUI COMMAND BassAmbientIntegration)
endif()
